var toast = document.getElementById('toast-1'); var thumb = toast.querySelector('.toast-thumb'); var iconOk = toast.querySelector('.toast-icon-success'); var iconError = toast.querySelector('.toast-icon-error'); var text = toast.querySelector('.toast-text'); var hideTimer = null; // -- Warteschlange: mehrere showToast()-Aufrufe kurz hintereinander (z. B. gestaute // Toast-Notes aus get_toast_notes()) würden sich sonst gegenseitig überschreiben, da es // nur ein gemeinsames Toast-Element gibt. Aufrufer müssen davon nichts wissen, die // öffentliche Signatur bleibt unverändert. var toastQueue = []; var toastShowing = false; // -- imgUrl ist optional (z. B. Produkt-Thumbnail beim "In den Warenkorb"); bestehende // Aufrufer ohne 4. Argument sind unverändert kompatibel. function showToast(ok, message, delay, imgUrl) { toastQueue.push({ ok: ok, message: message, delay: (typeof delay === 'number') ? delay : 4000, imgUrl: imgUrl || null }); if (!toastShowing) processToastQueue(); } function processToastQueue() { var next = toastQueue.shift(); if (!next) { toastShowing = false; return; } toastShowing = true; var hasThumb = !!next.imgUrl; thumb.classList.toggle('hidden', !hasThumb); if (hasThumb) thumb.src = next.imgUrl; iconOk.classList.toggle('hidden', hasThumb || !next.ok); iconError.classList.toggle('hidden', hasThumb || next.ok); toast.classList.toggle('bg-green-50', next.ok); toast.classList.toggle('border-green-200', next.ok); toast.classList.toggle('bg-red-50', !next.ok); toast.classList.toggle('border-red-200', !next.ok); text.innerHTML = next.message; text.classList.toggle('text-green-800', next.ok); text.classList.toggle('text-red-700', !next.ok); toast.classList.remove('hidden'); toast.classList.add('flex'); requestAnimationFrame(function () { toast.classList.remove('opacity-0', '-translate-y-2'); }); clearTimeout(hideTimer); hideTimer = setTimeout(function () { toast.classList.add('opacity-0', '-translate-y-2'); setTimeout(function () { toast.classList.add('hidden'); toast.classList.remove('flex'); processToastQueue(); }, 300); }, next.delay); }
