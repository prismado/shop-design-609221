% use utf8; // Produkt-Optionen (z. B. Farbe/Grösse) — Auswahl asynchron an den Server melden function setProductOption(selectEl, optId, prodId) { fetch('/dyn/set-product-option.html', { method: 'POST', body: new URLSearchParams({ opt_id: optId, id: prodId, opt: selectEl.value }) }) .then(function (res) { return res.json(); }) .then(function (data) { if (data.rc !== 0) return; refreshAssignedOptions(prodId); }) .catch(() => {}); } // Eine Produktoption entfernen, danach Tabelle + Zuschlagssumme neu laden function removeProductOption(prodId, optId) { fetch('/dyn/remove-product-option.html', { method: 'POST', body: new URLSearchParams({ prod_id: prodId, opt_id: optId }) }) .then(function () { refreshAssignedOptions(prodId); }) .catch(() => {}); } // Zentrale Quelle für Tabelle der zugewiesenen Optionen + Zuschlags-Summe // (aufgerufen beim initialen Laden, nach jeder Options-Änderung und nach jedem Entfernen) function refreshAssignedOptions(prodId) { var tbody = document.getElementById('assigned-options-body-' + prodId); var wrap = document.getElementById('assigned-options-wrap-' + prodId); var total = document.getElementById('opt_610041_179'); if (!tbody) return; fetch('/dyn/product-options-assigned.html?prod_id=' + encodeURIComponent(prodId)) .then(function (res) { return res.json(); }) .then(function (data) { if (data.rc !== 0) return; tbody.innerHTML = ''; var sum = 0; data.result.forEach(function (row) { sum += parseFloat(row.amount) || 0; // Zeile als Flexbox statt : Text (Label/Beschreibung) darf umbrechen/schrumpfen, // Betrag + Entfernen-Button bleiben als fixe Einheit immer sichtbar (auch auf schmalen Screens). var row_el = document.createElement('div'); row_el.className = 'flex items-start justify-between gap-3 p-3'; var textCol = document.createElement('div'); textCol.className = 'flex-1 min-w-0'; var lblEl = document.createElement('div'); lblEl.className = 'text-sm font-medium text-gray-800'; lblEl.textContent = row.lbl; var descEl = document.createElement('div'); descEl.className = 'text-xs text-gray-500'; descEl.textContent = row.desc; textCol.appendChild(lblEl); textCol.appendChild(descEl); var actionCol = document.createElement('div'); actionCol.className = 'flex items-center gap-2 flex-shrink-0'; var amountEl = document.createElement('span'); amountEl.className = 'text-sm font-semibold text-gray-800 whitespace-nowrap'; amountEl.textContent = 'CHF ' + (parseFloat(row.amount) || 0).toFixed(2); var removeLink = document.createElement('a'); removeLink.href = '#!'; removeLink.setAttribute('aria-label', 'Option entfernen'); removeLink.className = 'flex-shrink-0 inline-flex items-center justify-center w-6 h-6' + ' rounded-full italic font-serif text-lg leading-none text-gray-400 hover:text-gray-600' + ' hover:bg-gray-100 transition'; removeLink.textContent = '×'; removeLink.addEventListener('click', function (e) { e.preventDefault(); removeProductOption(prodId, row.opt_id); }); actionCol.appendChild(amountEl); actionCol.appendChild(removeLink); row_el.appendChild(textCol); row_el.appendChild(actionCol); tbody.appendChild(row_el); }); if (wrap) wrap.classList.toggle('hidden', data.result.length === 0); if (total) { total.textContent = sum.toFixed(2); } var surchargeLine = document.getElementById('options-surcharge-line-' + prodId); if (surchargeLine) { surchargeLine.classList.toggle('hidden', sum <= 0); } var surchargeHint = document.getElementById('surcharge-hint-610051'); if (surchargeHint) surchargeHint.classList.toggle('hidden', sum > 0); var basePriceEl = document.getElementById('base_price'); var mainPriceEl = document.getElementById('main-price-' + prodId); if (basePriceEl && mainPriceEl) { var basePrice = parseFloat(basePriceEl.textContent) || 0; mainPriceEl.textContent = 'CHF ' + (basePrice + sum).toFixed(2); } }) .catch(() => {}); }
