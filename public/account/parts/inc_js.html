% use utf8; // "Von Merkliste entfernen" (account/parts/main.html) — Seite wird nur einmal eingebunden, // daher liegt diese Funktion hier statt in der Wishlist-Karte selbst. function removeFromWishlist(el, prodId) { var body = new URLSearchParams(); body.set('prod_id', prodId); fetch('/dyn/remove-from-wishlist.html', { method: 'POST', body: body }) .then(function (res) { return res.json(); }) .then(function (data) { if (data.rc === 0) { var card = el.closest('.wishlist-item'); if (card) { card.style.transition = 'opacity .25s ease'; card.style.opacity = '0'; setTimeout(function () { card.remove(); }, 250); } var wishlistBadge = document.getElementById('wishlist-count'); if (wishlistBadge) { var n = Math.max(0, (parseInt(wishlistBadge.textContent, 10) || 0) - 1); wishlistBadge.textContent = n; wishlistBadge.classList.toggle('hidden', n <= 0); wishlistBadge.classList.toggle('flex', n > 0); } showToast(true, 'Artikel wurde von der Merkliste entfernt.'); } else { showToast(false, 'Fehler ' + data.rc + ': ' + data.message); } }) .catch(function () { showToast(false, 'Artikel konnte nicht entfernt werden.'); }); } // Beide Edit-Modals (Adresse / Konto, account/parts/main.html) teilen sich dieselbe Hülle // (id="edit-address") — nur Titel und nachgeladene URL unterscheiden sich, daher ein // gemeinsamer Loader statt doppeltem Fetch/Inject-Code. function loadEditModal(title, url) { var titleEl = document.getElementById('modal-title-edit-address'); var body = document.getElementById('modal-body-edit-address'); if (!body) return; if (titleEl) titleEl.textContent = title; body.innerHTML = '

Lädt…

'; openModal('edit-address'); fetch(url) .then(function (res) { return res.text(); }) .then(function (html) { body.innerHTML = html; }) .catch(function () { body.innerHTML = '

Formular konnte nicht geladen werden.

'; }); } // Adress-Bearbeiten-Modal — Inhalt hängt von der Adress-Type-ID des geklickten Stift-Buttons ab // (1=Rechnung, 2=Lieferung, weitere Kanäle möglich). function openEditAddressModal(addrTypeId) { loadEditModal('Adresse bearbeiten', '/account/modal-edit-address.html?addr_type_id=' + encodeURIComponent(addrTypeId)); } // Konto-/Benutzerinformation-Bearbeiten-Modal. function openEditAccountModal() { loadEditModal('Benutzerinformation bearbeiten', '/account/modal-edit-account.html'); } // Gemeinsame Submit-Logik für die Edit-Modals (Adresse / Konto) — beide dyn/*-Endpunkte liefern // dasselbe { rc, message }-Format und das Modal reagiert in beiden Fällen identisch. function submitEditModal(form, url, successMsg) { var body = new URLSearchParams(new FormData(form)); fetch(url, { method: 'POST', body: body }) .then(function (res) { return res.json(); }) .then(function (data) { if (data.rc === 0) { showToast(true, successMsg, 2000); closeModal('edit-address'); setTimeout(function () { window.location.href = window.location.pathname; }, 900); } else { showToast(false, data.message); } }) .catch(function () { showToast(false, 'Änderungen konnten nicht gespeichert werden.'); }); } // Speichern im Adress-Bearbeiten-Modal — POST an dyn/update-user-address.html. function saveAddress(form) { submitEditModal(form, '/dyn/update-user-address.html', 'Adresse wurde aktualisiert.'); } // Speichern im Konto-Bearbeiten-Modal — POST an dyn/update-user-account.html. function saveAccount(form) { submitEditModal(form, '/dyn/update-user-account.html', 'Benutzerinformation wurde aktualisiert.'); }
